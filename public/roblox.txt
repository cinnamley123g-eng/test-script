--================= 5. UTILITIES & COMPATIBILITY =================
local Utils = {}

function Utils.safeSetClipboard(text)
    pcall(function()
        if setclipboard then setclipboard(tostring(text))
        elseif toclipboard then toclipboard(tostring(text))
        elseif set_clipboard then set_clipboard(tostring(text))
        elseif Clipboard and Clipboard.set then Clipboard.set(tostring(text))
        end
    end)
end

function Utils.safeSetFPSCap(cap)
    pcall(function()
        if setfpscap then setfpscap(cap)
        elseif set_fps_cap then set_fps_cap(cap)
        end
    end)
end

function Utils.safeHttpRequest(url)
    local fn = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
    if fn then
        local success, res = pcall(function() return fn({ Url = url, Method = "GET" }) end)
        if success and res and res.Body then return res.Body end
    end
    if game.HttpGet then
        local success, res = pcall(function() return game:HttpGet(url) end)
        if success and res then return res end
    end
    return nil
end

function Utils.getHumanoid(targetChar)
    local char = targetChar or (player and player.Character)
    return char and char:FindFirstChildOfClass("Humanoid")
end

function Utils.getRootPart(targetChar)
    local char = targetChar or (player and player.Character)
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
end

function Utils.isPlayerAlive(p)
    if not p or not p.Parent or not p.Character then return false end
    local hum = p.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

function Utils.isTeammate(p)
    if not State.combat.teamCheck then return false end
    if not p or p == player then return true end

    if player.Team and p.Team then
        return player.Team == p.Team
    end
    if player.TeamColor and p.TeamColor and not player.Neutral and not p.Neutral then
        return player.TeamColor == p.TeamColor
    end

    local teamAttrKeys = { "Team", "TeamId", "Team_Id", "TeamName", "TeamColor", "Squad", "Faction", "Gang", "Crew", "Clan", "Party", "Side" }
    for _, key in ipairs(teamAttrKeys) do
        local myAttr = player:GetAttribute(key)
        local pAttr = p:GetAttribute(key)
        if myAttr ~= nil and pAttr ~= nil then
            return myAttr == pAttr
        end
    end

    local myChar = player.Character
    local pChar = p.Character
    if myChar and pChar then
        local myTeamVal = player:FindFirstChild("Team") or myChar:FindFirstChild("Team") or player:FindFirstChild("TeamId")
        local pTeamVal = p:FindFirstChild("Team") or pChar:FindFirstChild("Team") or p:FindFirstChild("TeamId")
        if myTeamVal and pTeamVal and myTeamVal:IsA("ValueBase") and pTeamVal:IsA("ValueBase") then
            return myTeamVal.Value == pTeamVal.Value
        end
    end

    return false
end

function Utils.isEnemy(p)
    if not p or p == player then return false end
    return not Utils.isTeammate(p)
end

function Utils.isPartVisible(targetPart)
    if not State.combat.wallCheck then return true end
    local cam = workspace.CurrentCamera
    if not cam or not targetPart then return false end

    local origin = cam.CFrame.Position
    local dir = (targetPart.Position - origin)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = RaycastFilterType.Exclude
    local ignoreList = { player.Character, cam }
    if targetPart.Parent then table.insert(ignoreList, targetPart.Parent) end
    rayParams.FilterDescendantsInstances = ignoreList

    local result = workspace:Raycast(origin, dir, rayParams)
    return result == nil
end

function UI.getSafeClampedPosition(targetPos, size, topExtra)
    local cam = workspace.CurrentCamera
    local viewport = cam and cam.ViewportSize or Vector2.new(1920, 1080)
    topExtra = topExtra or 50

    local w = size.X.Offset > 0 and size.X.Offset or (size.X.Scale * viewport.X)
    local h = size.Y.Offset > 0 and size.Y.Offset or (size.Y.Scale * viewport.Y)
    local halfW, halfH = w / 2, h / 2

    local minX, maxX = halfW + 10, viewport.X - halfW - 10
    local minY, maxY = halfH + topExtra + 10, viewport.Y - halfH - 10

    if minX > maxX then minX = viewport.X / 2 maxX = viewport.X / 2 end
    if minY > maxY then minY = viewport.Y / 2 maxY = viewport.Y / 2 end

    local currentX = targetPos.X.Scale * viewport.X + targetPos.X.Offset
    local currentY = targetPos.Y.Scale * viewport.Y + targetPos.Y.Offset

    local clampedX = math.clamp(currentX, minX, maxX)
    local clampedY = math.clamp(currentY, minY, maxY)

    return UDim2.new(0, clampedX, 0, clampedY)
end

function UI.makeDraggable(dragHandle, target, syncTarget, isMini, onClickCallback)
    local dragging = false
    local wasDragged = false
    local dragStart, startPos

    local bConn = dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if target:GetAttribute("IsMaximized") then return end
            dragging = true
            wasDragged = false
            dragStart = input.Position
            startPos = target.Position
            UI.bringToFront(target)
        end
    end)
    registerConn(bConn)

    local eConn = S.UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                if isMini and not wasDragged and onClickCallback then
                    onClickCallback()
                end
            end
        end
    end)
    registerConn(eConn)

    local cConn = S.UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then wasDragged = true end
            local rawPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            local clampedPos = UI.getSafeClampedPosition(rawPos, target.Size, isMini and 10 or 52)
            target.Position = clampedPos
            if syncTarget then syncTarget.Position = clampedPos end
        end
    end)
    registerConn(cConn)
end

local function closeWindow(id, animate)
    local frame = Tracker.openWindows[id]
    if not frame then return end
    Tracker.openWindows[id] = nil
    
    local mini = Tracker.miniTags[id]
    if mini then
        mini.Position = frame.Position
        mini.Visible = true
        local miniScale = mini:FindFirstChildOfClass("UIScale")
        if miniScale then
            miniScale.Scale = 0.5
            S.TweenS:Create(miniScale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end
    end

    if animate then
        local scale = frame:FindFirstChildOfClass("UIScale")
        if scale then
            S.TweenS:Create(scale, TweenInfo.new(0.15, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Scale = 0.6}):Play()
            task.delay(0.15, function() if frame.Parent and Tracker.openWindows[id] == nil then frame.Visible = false end end)
        else
            frame.Visible = false
        end
    else
        frame.Visible = false
    end
end

local function destroyWindow(id)
    if Tracker.windowFrames[id] then Tracker.windowFrames[id]:Destroy() end
    if Tracker.miniTags[id] then Tracker.miniTags[id]:Destroy() end
    Tracker.openWindows[id] = nil
    Tracker.windowFrames[id] = nil
    Tracker.miniTags[id] = nil
end

local function toggleWindow(def, contentFn, w, h)
    w, h = w or 410, h or 360

    if Tracker.openWindows[def.id] then
        UI.bringToFront(Tracker.openWindows[def.id])
        return
    end

    if Tracker.windowFrames[def.id] then
        local cached = Tracker.windowFrames[def.id]
        Tracker.openWindows[def.id] = cached
        cached.Visible = true
        cached.BackgroundTransparency = State.uiBackgroundTransp
        local tBar = cached:FindFirstChild("TitleBar")
        if tBar then tBar.BackgroundTransparency = State.titleBarTransp end

        if Tracker.miniTags[def.id] then Tracker.miniTags[def.id].Visible = false end
        cached.Position = UI.getSafeClampedPosition(cached.Position, cached.Size, 52)
        State.activeWindowId = def.id
        UI.bringToFront(cached)
        
        local scale = cached:FindFirstChildOfClass("UIScale")
        if scale then
            scale.Scale = 0.75
            S.TweenS:Create(scale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end
        return
    end

    Tracker.nextSpawnOffset = (Tracker.nextSpawnOffset + 35) % 110

    local win = Instance.new("Frame", windowLayer)
    win.Name             = "Win_" .. def.id
    win:SetAttribute("WindowId", def.id)
    win:SetAttribute("IsMaximized", false)
    win.AnchorPoint      = Vector2.new(0.5, 0.5)
    win.Size             = UDim2.new(0, w, 0, h)
    
    local initialPos = UDim2.new(0.5, Tracker.nextSpawnOffset - 20, 0.5, Tracker.nextSpawnOffset)
    win.Position         = UI.getSafeClampedPosition(initialPos, win.Size, 52)

    local winStroke = UI.mkStroke(def.color, 1.8, win, 0.15)
    Tracker.tabStrokes[def.id] = winStroke

    local defaultSize = win.Size
    local defaultPos = win.Position
    local winScale = Instance.new("UIScale", win)
    winScale.Scale = 0.75
    Tracker.windowFrames[def.id] = win

    local miniTag = Instance.new("TextButton", windowLayer)
    miniTag.Name = "Mini_" .. def.id
    miniTag.Size = UDim2.new(0, 46, 0, 46)
    miniTag.AnchorPoint = Vector2.new(0.5, 0.5)
    miniTag.Position = win.Position
    miniTag.BackgroundColor3 = Config.COL_PANEL
    miniTag.BackgroundTransparency = State.uiBackgroundTransp
    miniTag.Text = def.icon
    miniTag.TextColor3 = Config.COL_TEXT

        TextSize        = 13,
        TextColor3      = Config.COL_TEXT,
        TextXAlignment  = Enum.TextXAlignment.Left,
    }, titleBar)

    minBtn.Size             = UDim2.new(0, 24, 0, 24)
    minBtn.Position         = UDim2.new(1, -84, 0.5, -12)

    minBtn.Text             = "-"
    minBtn.TextColor3       = Config.COL_TEXT
    minBtn.Font             = Enum.Font.GothamBold

    UI.mkCorner(8, minBtn)

    maxBtn.Position         = UDim2.new(1, -56, 0.5, -12)

    maxBtn.BackgroundTransparency = 0.15

                    subTabButtons[i].BackgroundTransparency = (i == index) and 0 or 1
                    subTabButtons[i].TextColor3 = (i == index) and Config.COL_TEXT or Config.COL_SUBTEXT
                end
                State.activeWindowId = def.id
            end)
        end
    end

    for index = 1, #innerTabDefs do
        local contentBody = Instance.new("ScrollingFrame", win)
        contentBody.Size = UDim2.new(1, 0, 1, -bodyTopOffset)

        layout.Padding = UDim.new(0, 8)

        bodyPad.PaddingTop = UDim.new(0, 8)
        bodyPad.PaddingLeft = UDim.new(0, 12)
        bodyPad.PaddingRight = UDim.new(0, 12)
        bodyPad.PaddingBottom = UDim.new(0, 10)
        
        contentBodies[index] = contentBody
    end

    contentFn(contentBodies)
    
    for _, body in ipairs(contentBodies) do
        body.AutomaticCanvasSize = Enum.AutomaticSize.Y
    end

    Tracker.openWindows[def.id] = win
    State.activeWindowId = def.id
    UI.bringToFront(win)
    S.TweenS:Create(winScale, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()

function Controls.mkSliderRow(parent, icon, label, stateTbl, key, minVal, maxVal, defaultVal, suffix, accent, onChange)
    if stateTbl[key] == nil then stateTbl[key] = defaultVal end
    suffix = suffix or ""

    local row = Instance.new("Frame", parent)
    row.Size             = UDim2.new(1, 0, 0, 52)
    row.BackgroundColor3 = Config.COL_ROW
    row.BackgroundTransparency = 0.2
    UI.mkCorner(9, row)

    UI.mkLabel({
        Size = UDim2.new(0, 24, 0, 24), Position = UDim2.new(0, 10, 0, 6),
        Text = icon, TextSize = 14, TextColor3 = Config.COL_TEXT,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)

    UI.mkLabel({
        Size = UDim2.new(1, -95, 0, 24), Position = UDim2.new(0, 36, 0, 6),
        Text = label, TextSize = 12, Font = Enum.Font.GothamMedium,
        TextColor3 = Config.COL_TEXT, TextXAlignment = Enum.TextXAlignment.Left,
    }, row)

    local valLabel = UI.mkLabel({
        Size = UDim2.new(0, 55, 0, 20), Position = UDim2.new(1, -65, 0, 8),
        Text = tostring(stateTbl[key]) .. suffix, TextSize = 11, Font = Enum.Font.GothamBold,
        TextColor3 = accent or State.currentAccentColor, TextXAlignment = Enum.TextXAlignment.Right,
    }, row)

    local track = Instance.new("Frame", row)
    track.Size = UDim2.new(1, -20, 0, 6)
    track.Position = UDim2.new(0, 10, 0, 36)
    track.BackgroundColor3 = Config.COL_OFF
    UI.mkCorner(99, track)

    local fill = Instance.new("Frame", track)
    fill.Size = UDim2.new(math.clamp((stateTbl[key] - minVal) / (maxVal - minVal), 0, 1), 0, 1, 0)
    fill.BackgroundColor3 = accent or State.currentAccentColor
    UI.mkCorner(99, fill)

    local thumb = Instance.new("Frame", track)
    thumb.Size = UDim2.new(0, 14, 0, 14)
    thumb.AnchorPoint = Vector2.new(0.5, 0.5)
    thumb.Position = UDim2.new(math.clamp((stateTbl[key] - minVal) / (maxVal - minVal), 0, 1), 0, 0.5, 0)
    thumb.BackgroundColor3 = Config.COL_TEXT
    UI.mkCorner(99, thumb)

    local hit = Instance.new("TextButton", track)
    hit.Size = UDim2.new(1, 10, 1, 14)
    hit.Position = UDim2.new(0, -5, 0, -7)
    hit.BackgroundTransparency = 1
    hit.Text = ""

    local sliding = false
    local function updateFromInput(input)
        local relX = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(minVal + relX * (maxVal - minVal))
        stateTbl[key] = val
        valLabel.Text = tostring(val) .. suffix
        fill.Size = UDim2.new(relX, 0, 1, 0)
        thumb.Position = UDim2.new(relX, 0, 0.5, 0)
        if onChange then onChange(val) end
    end

    local hConn = hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            updateFromInput(input)
        end
    end)
    registerConn(hConn)

    local uEndConn = S.UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)
    registerConn(uEndConn)

    local uChgConn = S.UIS.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    registerConn(uChgConn)
end

local flyingBodyVelocity, flyingBodyGyro
function Logic.setFly(enable)
    local rootPart = Utils.getRootPart()
    local hum = Utils.getHumanoid()
    if enable and rootPart then
        if hum then hum.PlatformStand = true end
        if not flyingBodyVelocity then
            flyingBodyVelocity = Instance.new("BodyVelocity")
            flyingBodyVelocity.Velocity = Vector3.zero
            flyingBodyVelocity.MaxForce = Vector3.new(1e6, 1e6, 1e6)
            flyingBodyVelocity.Parent = rootPart
            table.insert(Tracker.spawned, flyingBodyVelocity)
        end
        if not flyingBodyGyro then
            flyingBodyGyro = Instance.new("BodyGyro")
            flyingBodyGyro.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
            flyingBodyGyro.CFrame = rootPart.CFrame
            flyingBodyGyro.Parent = rootPart
            table.insert(Tracker.spawned, flyingBodyGyro)
        end
    else
        if flyingBodyVelocity then flyingBodyVelocity:Destroy() flyingBodyVelocity = nil end
        if flyingBodyGyro then flyingBodyGyro:Destroy() flyingBodyGyro = nil end
        if hum then
            hum.PlatformStand = false
            hum:ChangeState(Enum.HumanoidStateType.Freefall)
        end
    end
end

local function getBestAimbotTarget()
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local mousePos = S.UIS:GetMouseLocation()
    local bestTarget = nil
    local bestDist = State.combat.aimFOV

    for _, p in ipairs(S.Players:GetPlayers()) do
        if p ~= player and Utils.isPlayerAlive(p) and not Utils.isTeammate(p) then
            local char = p.Character
            local targetPart = char and (char:FindFirstChild(State.combat.aimTargetPart) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"))
            if targetPart and Utils.isPartVisible(targetPart) then
                local screenPos, onScreen = cam:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist2D <= bestDist then
                        bestDist = dist2D
                        bestTarget = { player = p, part = targetPart }
                    end
                end
            end
        end
    end
    return bestTarget
end

    if State.target.loopOrbit then
        State.target.orbitAngle = (State.target.orbitAngle + State.target.orbitSpeed * dt) % (math.pi * 2)
        local offset = Vector3.new(math.cos(State.target.orbitAngle) * State.target.orbitRadius, 1.5, math.sin(State.target.orbitAngle) * State.target.orbitRadius)
        myRoot.CFrame = CFrame.new(tRoot.Position + offset, tRoot.Position)
    elseif State.target.loopFollow then
        local dist = (myRoot.Position - tRoot.Position).Magnitude
        if dist > State.target.followDistance then
            local targetPos = tRoot.Position - (tRoot.CFrame.LookVector * State.target.followDistance)
            myRoot.CFrame = myRoot.CFrame:Lerp(CFrame.new(targetPos, tRoot.Position), 0.1)
        end
    end
end)

--================= 14. SERVER & OPTIMIZATION TOOLS =================
local Server = {}

function Server.rejoin()
    pcall(function()
        S.TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, player)
    end)
end

function Server.serverHop()
    pcall(function()
        local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local body = Utils.safeHttpRequest(url)
        if body then
            local success, result = pcall(function() return S.HttpService:JSONDecode(body) end)
            if success and result and result.data then
                local serverList = {}
                for _, s in ipairs(result.data) do
                    if s.playing and s.maxPlayers and s.playing < s.maxPlayers and s.id ~= game.JobId then
                        table.insert(serverList, s.id)
                    end
                end
                if #serverList > 0 then
                    local chosen = serverList[math.random(1, #serverList)]
                    S.TeleportService:TeleportToPlaceInstance(game.PlaceId, chosen, player)
                    return
                end
            end
        end
        S.TeleportService:Teleport(game.PlaceId, player)
    end)
end

function Server.boostFPS()
    pcall(function()
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") and not v:IsA("MeshPart") and not v:IsA("Terrain") then
                v.Material = Enum.Material.SmoothPlastic
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = 1
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") then
                v.Enabled = false
            end
        end
        S.Lighting.GlobalShadows = false
        S.Lighting.FogEnd = 9e9
    end)
end
