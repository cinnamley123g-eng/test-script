local Utils = {}

function Utils.safeSetClipboard(text)
    pcall(function()
        if setclipboard then
            setclipboard(tostring(text))
        elseif toclipboard then
            toclipboard(tostring(text))
        elseif set_clipboard then
            set_clipboard(tostring(text))
        elseif Clipboard and Clipboard.set then
            Clipboard.set(tostring(text))
        end
    end)
end

function Utils.safeSetFPSCap(cap)
    pcall(function()
        if setfpscap then
            setfpscap(cap)
        elseif set_fps_cap then
            set_fps_cap(cap)
        end
    end)
end

function Utils.safeHttpRequest(url)
    local fn = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
    if fn then
        local success, res = pcall(function()
            return fn({ Url = url, Method = "GET" })
        end)
        if success and res and res.Body then
            return res.Body
        end
    end
    if game.HttpGet then
        local success, res = pcall(function()
            return game:HttpGet(url)
        end)
        if success and res then
            return res
        end
    end
    return nil
end

function Utils.getHumanoid()
    local char = player.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

function Utils.getRootPart()
    local char = player.Character
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
end

function Utils.isPlayerAlive(p)
    if not p or not p.Parent or not p.Character then return false end
    local hum = p.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

function Utils.isTeammate(p)
    if not State.combat.teamCheck then return false end
    if not p or p == player then return true end

    if player.Team and p.Team then
        return player.Team == p.Team
    end

    if player.TeamColor and p.TeamColor and not player.Neutral and not p.Neutral then
        return player.TeamColor == p.TeamColor
    end

    local teamAttrKeys = { "Team", "TeamId", "Team_Id", "TeamName", "TeamColor", "Squad", "Faction", "Gang", "Crew", "Clan", "Party", "Side" }
    for _, key in ipairs(teamAttrKeys) do
        local myAttr = player:GetAttribute(key)
        local pAttr = p:GetAttribute(key)
        if myAttr ~= nil and pAttr ~= nil then
            return myAttr == pAttr
        end
    end

    local myChar = player.Character
    local pChar = p.Character
    if myChar and pChar then
        for _, key in ipairs(teamAttrKeys) do
            local myCharAttr = myChar:GetAttribute(key)
            local pCharAttr = pChar:GetAttribute(key)
            if myCharAttr ~= nil and pCharAttr ~= nil then
                return myCharAttr == pCharAttr
            end
        end

        local myTeamVal = player:FindFirstChild("Team") or myChar:FindFirstChild("Team") or player:FindFirstChild("TeamId")
        local pTeamVal = p:FindFirstChild("Team") or pChar:FindFirstChild("Team") or p:FindFirstChild("TeamId")
        if myTeamVal and pTeamVal and myTeamVal:IsA("ValueBase") and pTeamVal:IsA("ValueBase") then
            return myTeamVal.Value == pTeamVal.Value
        end
    end

    return false
end

function Utils.isEnemy(p)
    if not p or p == player then return false end
    return not Utils.isTeammate(p)
end

function UI.bringToFront(frame)
    if not frame then return end
    Tracker.zCounter += 10
    frame.ZIndex = Tracker.zCounter
    local winId = frame:GetAttribute("WindowId")
    if winId then
        local mini = Tracker.miniTags[winId]
        if mini then mini.ZIndex = Tracker.zCounter + 2 end
    end
end

function UI.getSafeClampedPosition(targetPos, size, topExtra)
    local cam = workspace.CurrentCamera
    local viewport = cam and cam.ViewportSize or Vector2.new(1920, 1080)
    topExtra = topExtra or 50

    local w = size.X.Offset > 0 and size.X.Offset or (size.X.Scale * viewport.X)
    local h = size.Y.Offset > 0 and size.Y.Offset or (size.Y.Scale * viewport.Y)

    local halfW = w / 2
    local halfH = h / 2

    local minX = halfW + 10
    local maxX = viewport.X - halfW - 10
    local minY = halfH + topExtra + 10
    local maxY = viewport.Y - halfH - 10

    if minX > maxX then minX = viewport.X / 2 maxX = viewport.X / 2 end
    if minY > maxY then minY = viewport.Y / 2 maxY = viewport.Y / 2 end

    local currentX = targetPos.X.Scale * viewport.X + targetPos.X.Offset
    local currentY = targetPos.Y.Scale * viewport.Y + targetPos.Y.Offset

    local clampedX = math.clamp(currentX, minX, maxX)
    local clampedY = math.clamp(currentY, minY, maxY)

    return UDim2.new(0, clampedX, 0, clampedY)
end

function UI.makeDraggable(dragHandle, target, syncTarget, isMini, onClickCallback)
    local dragging = false
    local wasDragged = false
    local dragStart, startPos

    local bConn = dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if target:GetAttribute("IsMaximized") then return end
            dragging = true
            wasDragged = false
            dragStart = input.Position
            startPos = target.Position
            UI.bringToFront(target)
        end
    end)
    registerConn(bConn)

    local eConn = S.UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                if isMini and not wasDragged and onClickCallback then
                    onClickCallback()
                end
            end
        end
    end)
    registerConn(eConn)

    local cConn = S.UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then
                wasDragged = true
            end
            local rawPos = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
            local clampedPos = UI.getSafeClampedPosition(rawPos, target.Size, isMini and 10 or 52)
            target.Position = clampedPos
            if syncTarget then
                syncTarget.Position = clampedPos
            end
        end
    end)
    registerConn(cConn)
end

--================= WINDOW MANAGER =================
local function closeWindow(id, animate)
    local frame = Tracker.openWindows[id]
    if not frame then return end
    Tracker.openWindows[id] = nil
    
    local mini = Tracker.miniTags[id]
    if mini then
        mini.Position = frame.Position
        mini.Visible = true
        local miniScale = mini:FindFirstChildOfClass("UIScale")
        if miniScale then
            miniScale.Scale = 0.5
            S.TweenS:Create(miniScale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end
    end

    if animate then
        local scale = frame:FindFirstChildOfClass("UIScale")
        if scale then
            S.TweenS:Create(scale, TweenInfo.new(0.15, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Scale = 0.6}):Play()
            task.delay(0.15, function() 
                if frame.Parent and Tracker.openWindows[id] == nil then 
                    frame.Visible = false 
                end 
            end)
        else
            frame.Visible = false
        end
    else
        frame.Visible = false
    end
end

local function destroyWindow(id)
    if Tracker.windowFrames[id] then Tracker.windowFrames[id]:Destroy() end
    if Tracker.miniTags[id] then Tracker.miniTags[id]:Destroy() end
    Tracker.openWindows[id] = nil
    Tracker.windowFrames[id] = nil
    Tracker.miniTags[id] = nil
end

    if Tracker.windowFrames[def.id] then
        local cached = Tracker.windowFrames[def.id]
        Tracker.openWindows[def.id] = cached
        cached.Visible = true
        cached.BackgroundTransparency = State.uiBackgroundTransp
        local tBar = cached:FindFirstChildOfClass("Frame")
        if tBar then
            tBar.BackgroundTransparency = State.titleBarTransp

        cached.Position = UI.getSafeClampedPosition(cached.Position, cached.Size, 52)

local hudConn = S.RunS.RenderStepped:Connect(function()
    frameCount += 1
    local now = tick()
    if now - lastFpsTime >= 0.5 then
        local fps = math.floor(frameCount / (now - lastFpsTime))
        frameCount = 0
        lastFpsTime = now
        local ping = 35
        pcall(function()
            local net = S.StatsService and S.StatsService.Network
            local sPing = net and net.ServerStatsItem and net.ServerStatsItem["Data Ping"]
            if sPing then ping = math.floor(sPing:GetValue()) end
        end)
        local memMb = 120
        pcall(function()
            if S.StatsService and S.StatsService.GetTotalMemoryUsageMb then
                memMb = math.floor(S.StatsService:GetTotalMemoryUsageMb())
            else
                memMb = math.floor(collectgarbage("count") / 1024)
            end
        end)
        hudLabel.Text = string.format("⚡ FPS: %d  |  📶 Ping: %dms  |  💾 %dMB", math.clamp(fps, 1, 999), math.clamp(ping, 0, 999), memMb)
    end
end)

function Controls.mkSliderRow(parent, icon, label, stateTbl, key, minVal, maxVal, defaultVal, suffix, accent, onChange)

    UI.mkCorner(99, track)

    local fill = Instance.new("Frame", track)
    fill.Size = UDim2.new(math.clamp((stateTbl[key] - minVal) / (maxVal - minVal), 0, 1), 0, 1, 0)
    fill.BackgroundColor3 = accent or State.currentAccentColor
    UI.mkCorner(99, fill)

    local thumb = Instance.new("Frame", track)
    thumb.Size = UDim2.new(0, 14, 0, 14)
    thumb.AnchorPoint = Vector2.new(0.5, 0.5)
    thumb.Position = UDim2.new(math.clamp((stateTbl[key] - minVal) / (maxVal - minVal), 0, 1), 0, 0.5, 0)
    thumb.BackgroundColor3 = Config.COL_TEXT
    UI.mkCorner(99, thumb)

    hit.Position = UDim2.new(0, -5, 0, -7)
    hit.BackgroundTransparency = 1
    hit.Text = ""

    local sliding = false
    local function updateFromInput(input)
        local relX = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = math.floor(minVal + relX * (maxVal - minVal))
        stateTbl[key] = val
        valLabel.Text = tostring(val) .. suffix
        fill.Size = UDim2.new(relX, 0, 1, 0)
        thumb.Position = UDim2.new(relX, 0, 0.5, 0)
        if onChange then onChange(val) end
    end

    local hConn = hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            updateFromInput(input)
        end
    end)
    registerConn(hConn)

    local uEndConn = S.UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)
    registerConn(uEndConn)

    local uChgConn = S.UIS.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    registerConn(uChgConn)
end

--================= REAL LOGIC IMPLEMENTATIONS & BUG FIXES =================
local Logic = {}

-- 1. FIXED TPWALK (CFrame Speed Bypass without rubberbanding)
local tpwalkConn = S.RunS.Heartbeat:Connect(function(dt)
    if State.player.tpwalk and not State.player.fly then
        local hum = Utils.getHumanoid()
        local rootPart = Utils.getRootPart()
        if hum and rootPart and hum.MoveDirection.Magnitude > 0 then
            local speedMultiplier = (State.player.tpwalkSpeed or 2) * 22 * dt
            local moveVec = hum.MoveDirection * speedMultiplier
            -- Clamp vertical drift to avoid gravity fighting
            rootPart.AssemblyLinearVelocity = Vector3.new(0, rootPart.AssemblyLinearVelocity.Y, 0)
            rootPart.CFrame = rootPart.CFrame + moveVec
        end
    end
end)
registerConn(tpwalkConn)

-- 2. FIXED INFINITE JUMP (Reliable Velocity Impulse in Mid-Air)
local infJumpConn = S.UIS.JumpRequest:Connect(function()
    if State.player.infjump then
        local hum = Utils.getHumanoid()
        local rootPart = Utils.getRootPart()
        if hum and rootPart and hum.Health > 0 then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
            local jPower = hum.JumpPower > 0 and hum.JumpPower or 50
            rootPart.AssemblyLinearVelocity = Vector3.new(rootPart.AssemblyLinearVelocity.X, jPower, rootPart.AssemblyLinearVelocity.Z)
        end
    end
end)
registerConn(infJumpConn)

-- 3. FIXED GHOST MODE (Chế độ ma: Semi-transparency, Noclip & Collision Filter)
function Logic.setGhostMode(enable)
    local char = player.Character
    if not char then return end

    if enable then
        State.player.ghostOriginalTransp = {}
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                State.player.ghostOriginalTransp[part] = part.Transparency
                part.Transparency = 0.65
                part.CanCollide = false
            elseif part:IsA("Decal") then
                State.player.ghostOriginalTransp[part] = part.Transparency
                part.Transparency = 0.7
            end
        end
    else
        for part, orig in pairs(State.player.ghostOriginalTransp) do
            if part and part.Parent then
                pcall(function()
                    part.Transparency = orig
                    if part:IsA("BasePart") and (part.Name == "Head" or part.Name:find("Torso") or part.Name:find("Leg")) then
                        part.CanCollide = true
                    end
                end)
            end
        end
        State.player.ghostOriginalTransp = {}
    end
end

local ghostLoop = S.RunS.Stepped:Connect(function()
    if State.player.ghost or State.player.noclip then
        local char = player.Character
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end
end)
registerConn(ghostLoop)

-- 4. FIXED HEAD SIT (Seamless CFrame Head Lock without Physics Desync)
local headSitConn = S.RunS.Heartbeat:Connect(function()
    if State.target.loopHeadSit and State.target.selectedPlayer then
        local myRoot = Utils.getRootPart()
        local myHum = Utils.getHumanoid()
        local tChar = State.target.selectedPlayer.Character
        local tHead = tChar and (tChar:FindFirstChild("Head") or tChar:FindFirstChild("HumanoidRootPart"))
        if myRoot and myHum and tHead then
            myHum.Sit = true
            myRoot.AssemblyLinearVelocity = Vector3.zero
            myRoot.CFrame = tHead.CFrame * CFrame.new(0, 1.8, 0)
        end
    end
end)
registerConn(headSitConn)

-- Fly System
local flyingBodyVelocity, flyingBodyGyro
function Logic.setFly(enable)
    local rootPart = Utils.getRootPart()
    local hum = Utils.getHumanoid()
    if enable and rootPart then
        if hum then hum.PlatformStand = true end
        if not flyingBodyVelocity then
            flyingBodyVelocity = Instance.new("BodyVelocity")
            flyingBodyVelocity.Velocity = Vector3.zero
            flyingBodyVelocity.MaxForce = Vector3.new(1e6, 1e6, 1e6)
            flyingBodyVelocity.Parent = rootPart
            table.insert(Tracker.spawned, flyingBodyVelocity)
        end
        if not flyingBodyGyro then
            flyingBodyGyro = Instance.new("BodyGyro")
            flyingBodyGyro.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
            flyingBodyGyro.CFrame = rootPart.CFrame
            flyingBodyGyro.Parent = rootPart
            table.insert(Tracker.spawned, flyingBodyGyro)
        end
    else
        if flyingBodyVelocity then flyingBodyVelocity:Destroy() flyingBodyVelocity = nil end
        if flyingBodyGyro then flyingBodyGyro:Destroy() flyingBodyGyro = nil end
        if hum then
            hum.PlatformStand = false
            hum:ChangeState(Enum.HumanoidStateType.Freefall)
        end
    end
end

local flyLoopConn = S.RunS.RenderStepped:Connect(function()
    if State.player.fly and flyingBodyVelocity and flyingBodyGyro then
        local rootPart = Utils.getRootPart()
        local cam = workspace.CurrentCamera
        local hum = Utils.getHumanoid()
        if rootPart and cam and hum then
            hum.PlatformStand = true
            local moveDir = Vector3.zero
            if S.UIS:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
            if S.UIS:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
            if S.UIS:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
            if S.UIS:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
            if S.UIS:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
            if S.UIS:IsKeyDown(Enum.KeyCode.LeftShift) or S.UIS:IsKeyDown(Enum.KeyCode.LeftControl) then moveDir = moveDir - Vector3.new(0, 1, 0) end

            if moveDir.Magnitude > 0 then
                flyingBodyVelocity.Velocity = moveDir.Unit * State.player.flySpeed
            else
                flyingBodyVelocity.Velocity = Vector3.zero
            end
            flyingBodyGyro.CFrame = cam.CFrame
        end
    end
end)
registerConn(flyLoopConn)

-- Server Tools (5. FIXED REGISTER OVERFLOW for serverHop)
local Server = {}
function Server.rejoin()
    pcall(function()
        S.TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, player)
    end)
end

function Server.serverHop()
    pcall(function()
        local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local body = Utils.safeHttpRequest(url)
        if body then
            local success, result = pcall(function()
                return S.HttpService:JSONDecode(body)
            end)
            if success and result and result.data then
                local serverList = {}
                for _, s in ipairs(result.data) do
                    if s.playing and s.maxPlayers and s.playing < s.maxPlayers and s.id ~= game.JobId then
                        table.insert(serverList, s.id)
                    end
                end
                if #serverList > 0 then
                    local chosen = serverList[math.random(1, #serverList)]
                    S.TeleportService:TeleportToPlaceInstance(game.PlaceId, chosen, player)
                    return
                end
            end
        end
        S.TeleportService:Teleport(game.PlaceId, player)
    end)
end

function Server.boostFPS()
    pcall(function()
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") and not v:IsA("MeshPart") and not v:IsA("Terrain") then
                v.Material = Enum.Material.SmoothPlastic
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = 1
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") then
                v.Enabled = false
            end
        end
        S.Lighting.GlobalShadows = false
        S.Lighting.FogEnd = 9e9
    end)
end
